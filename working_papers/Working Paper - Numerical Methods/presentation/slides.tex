\section{The numerical question}
\begin{frame}{What problem does this paper solve?}
\begin{block}{The computational task}
For a fixed difference order $d$, rolling window $L$, and forecast horizon $h$,
find and compare the relevant minima of a forecast-validation loss
$F_{d,L,h}(S)$ over $S\in[0,1]$.
\end{block}
\begin{columns}[T,totalwidth=\linewidth]
\begin{column}{.49\linewidth}
\textbf{Established ingredients}
\begin{itemize}
\item Finite-difference penalized least squares.
\item Chronological forecast-validation loss.
\item Scalar numerical search and root finding.
\end{itemize}
\end{column}
\begin{column}{.49\linewidth}
\textbf{Our numerical focus}
\begin{itemize}
\item Multimodal loss geometry.
\item Exact limiting endpoints.
\item Adaptive stationary-point discovery.
\item Finite rational root isolation.
\end{itemize}
\end{column}
\end{columns}
\SourceLine{Not a new PLS smoother, not the first forecast-based CV criterion, and not a claim of predictive superiority.}
\end{frame}

\begin{frame}{Model and objective: all numerical claims are conditional}
\[
\widehat{\tau}_{\lambda} =
\underset{\tau}{\arg\min}\left\{\lVert y-\tau\rVert_2^2+
\lambda\lVert D_d\tau\rVert_2^2\right\}
=H_\lambda y,\qquad
H_\lambda=(I+\lambda D_d^\top D_d)^{-1}.
\]
\[
F_{d,L,h}(S)=\frac{1}{|\mathcal O|}
\sum_{T\in\mathcal O}\frac{1}{h}
\left\|z_T-G_{d,h}H_{\lambda(S)}x_T\right\|_2^2.
\]
\begin{block}{Scope}
Identity weighting $V=I$, zero penalized mean difference $\mu=0$, regularly
spaced observations, fixed $(d,L,h)$, and native zero-$d$-th-difference continuation.
\end{block}
\SourceLine{Each rolling-origin loss uses training data available at that origin; validation targets are observed future blocks.}
\end{frame}

\section{Conclusions}
\begin{frame}{What have we established?}
\begin{enumerate}
\setlength{\itemsep}{.45em}
\item The forecast-validation objective for finite-difference PLS can be \textbf{multimodal}; both endpoints are meaningful models.
\item A derivative-aware, boundary-aware adaptive search recovered the finite-reference minima in the tested synthetic and financial suites at \textbf{1.57--1.84\%} of dense-grid evaluation counts.
\item A distinct \textbf{Sturm construction} counts every positive stationary root for a specified small rational instance; it is not a scalable completeness guarantee.
\item Multiple candidate smoothness values can imply \textbf{different fitted trends and future paths}, but the validation winner need not remain the test winner.
\end{enumerate}
\SourceLine{All numerical results refer to the documented frozen configurations, data, and comparison protocols.}
\end{frame}

\begin{frame}{Boundaries of the contribution}
\begin{columns}[T,totalwidth=\linewidth]
\begin{column}{.49\linewidth}
\begin{block}{Supported}
\begin{itemize}
\item Analytic derivatives for the fixed continuation model.
\item Empirical dense-reference recovery on tested surfaces.
\item Exact limiting endpoint treatment.
\item Sturm certification of stationary-root counts in finite rational cases.
\end{itemize}
\end{block}
\end{column}
\begin{column}{.49\linewidth}
\begin{block}{Not established}
\begin{itemize}
\item Universal recovery by finite adaptive sampling.
\item Global certification of numeric objective values for large windows.
\item Uniform improvement in external forecast accuracy.
\item Financial predictability or trading value.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\SourceLine{Dense-grid agreement is a numerical benchmark, not a theorem of exhaustive root discovery.}
\end{frame}
